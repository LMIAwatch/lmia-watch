# LMIA Watch. Every page is open to search engines, in English and in French
# (?lang=fr). The payment webhook, the JSON API, the bulk downloads and the
# response form are not pages to index.
User-agent: *
Allow: /
Disallow: /hooks/
Disallow: /api/
Disallow: /downloads/
Disallow: /employers/*/respond
# A search with filters or a page cursor, a correction form opened from a
# profile (?employer=) and a map opened on a postal area (?fsa=) each name the
# plain page as their canonical URL, so crawlers need not walk the query
# forms. The plain pages and their French versions (?lang=fr) stay open.
Disallow: /search?
Allow: /search?lang=fr$
Disallow: /corrections?
Allow: /corrections?lang=fr$
Disallow: /map?
Allow: /map?lang=fr$
Crawl-delay: 5

# Bing honours Crawl-delay, and five seconds would cap it at about 17,000
# pages a day against 160,000 employer pages. A crawler reads only the group
# that names it, so Bingbot's group repeats the rules without the delay.
User-agent: bingbot
Allow: /
Disallow: /hooks/
Disallow: /api/
Disallow: /downloads/
Disallow: /employers/*/respond
Disallow: /search?
Allow: /search?lang=fr$
Disallow: /corrections?
Allow: /corrections?lang=fr$
Disallow: /map?
Allow: /map?lang=fr$
